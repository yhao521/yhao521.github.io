export RESTART_TOMCAT_HOME=${BASEPATH}/cloud-tomcat-8880

echo "[Deploy] Shutting down  tomcat"
PIDS=`ps -f | grep java | grep "${RESTART_TOMCAT_HOME}" |awk '{print $2}'`
for PID in $PIDS ; do
    kill $PID > /dev/null 2>&1
done
COUNT=0
while [ $COUNT -lt 1 ]; do
    echo -e ".\c"
    sleep 1
    COUNT=1
    for PID in $PIDS ; do
        PID_EXIST=`ps -f -p $PID | grep java`
        if [ -n "$PID_EXIST" ]; then
            COUNT=0
            break
        fi
    done
done

echo "[Deploy] Shutted down  tomcat PID:${PIDS}"

echo "[Clean] Cleaning up  tomcat logs"
rm -rf $RESTART_TOMCAT_HOME/logs/*

忘记了admin管理员的账号密码

如果一不小心忘记了admin管理员的账号密码。然后就看不到“管理用户”那部分内容了，看不到就改不了用户权限，也就是系统瘫痪了。
这个时候该怎么办？？？？

别慌，静下来回想下，在jenkins环境刚搭建好并没注册账号和密码的时候，都能看见，也就是没有sign in 和sign up模块的时候，是可以使用的，这些信息应该写在jenkins的配置文件里。

路径在…/jenkins/config.xml

修复办法：千万注意：修复前一定要提前备份config.xml文件

然后需要删除config.xml文件中的以下内容：
