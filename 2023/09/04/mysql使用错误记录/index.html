gtid跳过错误：

stop slave ;
SET @@SESSION.GTID_NEXT= 'dfb5c289-3e46-11e9-988a-fa163ed1a4a7:154212';
BEGIN; COMMIT;
SET SESSION GTID_NEXT = AUTOMATIC;
start slave;



SET @@SESSION.GTID_NEXT= 'dfb5c289-3e46-11e9-988a-fa163ed1a4a7:154168';
BEGIN; COMMIT;
SET SESSION GTID_NEXT = AUTOMATIC;



stop slave ;
SET @@SESSION.GTID_NEXT= 'dfb5c289-3e46-11e9-988a-fa163ed1a4a7:154169';
BEGIN; COMMIT;
SET SESSION GTID_NEXT = AUTOMATIC;
start slave;
show slave status\G

stop slave ;
SET @@SESSION.GTID_NEXT= 'dfb5c289-3e46-11e9-988a-fa163ed1a4a7:154170';
BEGIN; COMMIT;
SET SESSION GTID_NEXT = AUTOMATIC;
start slave;
show slave status\G


stop slave ;
SET @@SESSION.GTID_NEXT= 'dfb5c289-3e46-11e9-988a-fa163ed1a4a7:154171';
BEGIN; COMMIT;
SET SESSION GTID_NEXT = AUTOMATIC;
start slave;
show slave status\G

设置GTID主从：

stop slave ;
change master to master_host='172.18.231.121',master_port=6660,master_user='mysql', master_password='mysql', master_auto_position=1;
start slave;
show slave status\G

忽略某张表：

CHANGE REPLICATION FILTER Replicate_Wild_Ignore_Table = ('');
