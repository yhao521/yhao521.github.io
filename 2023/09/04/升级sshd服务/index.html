下载

wget https://cdn.openbsd.org/pub/OpenBSD/OpenSSH/portable/openssh-8.4p1.tar.gz

tar -zxvf openssh-8.4p1.tar.gz

cd openssh-8.4p1

./configure –prefix=/usr/ –sysconfdir=/etc/ssh/ –with-ssl –with-md5-passwords mandir=/usr/share/man/

make && make install

缺少依赖： yum install -y openssl-devel zlib-devel gcc

删除/etc/ssh/下的密钥对，rm -f /etc/ssh/sshhost* （只删除密钥对即可，在重启的时候会重新生成）

复制启动文件至/etc/init.d/

cp contrib/redhat/sshd.init /etc/init.d/sshd

修改启动文件：vim /etc/init.d/sshd，将PID_FILE路径改为原来的sshd.pid的路径

[root@myhost7 openssh-8.4p1]# /etc/init.d/sshd stop
Stopping sshd (via systemctl): [ OK ]
[root@myhost7 openssh-8.4p1]# /etc/init.d/sshd start
Starting sshd (via systemctl): [ OK ]

1.在服务器上的随意目录里下载SSH安装包，如不知道去什么地方下载，可以用浏览器百度SSH官网进行下载；

2.将下载好的SSH包，进行解压，并进入解压后生成的目录：

# tar -xf openssh-8.0p1.tar.gz
# cd openssh-8.0p1

3.进行编译安装：

# ./configure –prefix=/usr –sysconfdir=/etc/ssh –with-md5-passwords –with-pam –with-zlib –with-openssl-includes=/usr –with-privsep-path=/var/lib/sshd

# make && make install

4.将新的配置文件添加到指定的SSH目录，并查看SSH版本：

# install -v -m755 contrib/ssh-copy-id /usr/bin
# install -v -m644 contrib/ssh-copy-id.1 /usr/share/man/man1
# install -v -m755 -d /usr/share/doc/openssh-8.0p1
# install -v -m644 INSTALL LICENCE OVERVIEW README* /usr/share/doc/openssh-8.0p1
# ssh -V

5.配置并重启：

echo “PermitRootLogin yes” >> /etc/ssh/sshd_config
cp -p contrib/redhat/sshd.init /etc/init.d/sshd
chmod +x /etc/init.d/sshd
chkconfig –add sshd
chkconfig sshd on
/etc/init.d/sshd restart
