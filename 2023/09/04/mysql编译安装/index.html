mysql-5.7编译安装

下载安装包 https://downloads.mysql.com/archives/community/

设置预编译命令

cmake . \

-DCMAKE_INSTALL_PREFIX=/mysql/mysql-5.7 \

-DDEFAULT_CHARSET=utf8 \

-DDEFAULT_COLLATION=utf8_general_ci \

-DMYSQL_DATADIR=/data/3336 \

-DMYSQL_USER=mysql \

-DMYSQL_TCP_PORT=3336 \

-DSYSCONFDIR=/mysql/mysql-5.7/etc \

-DWITH_BOOST=boost

样例：

cmake . -DWITH_BOOST=boost/boost_1_59_0/ -DCMAKE_INSTALL_PREFIX=/usr/local/mysql -DSYSCONFDIR=/etc -DMYSQL_DATADIR=/usr/local/mysql/data -DINSTALL_MANDIR=/usr/share/man -DMYSQL_TCP_PORT=3306 -DMYSQL_UNIX_ADDR=/tmp/mysql.sock -DDEFAULT_CHARSET=utf8 -DEXTRA_CHARSETS=all -DDEFAULT_COLLATION=utf8_general_ci -DWITH_READLINE=1 -DWITH_SSL=system -DWITH_EMBEDDED_SERVER=1 -DENABLED_LOCAL_INFILE=1 -DWITH_INNOBASE_STORAGE_ENGINE=1

make && make install

初始化数据库

cd /mysql/mysql-5.7/bin;

./mysqld –initialize –user=mysql –basedir=/mysql/mysql-5.7 –datadir=/data/3336 –innodb_undo_tablespaces=3 –explicit_defaults_for_timestamp

temporary password is generated for root@localhost: st5iMcPFby;O

启动

cd /mysql/mysql-5.7;./bin/mysqld_safe –defaults-file=/mysql/mysql-5.7/etc/my3336.cnf –user=mysql &

cd /mysql/mysql-5.7;./bin/mysqladmin –defaults-file=/mysql/mysql-5.7/etc/my3336.cnf –socket=/data/3336/mysql.sock -umysql -pmysql shutdown

登录

./bin/mysql -P3336 -uroot –socket=/data/3336/mysql.sock -p

修改密码

set password=password(“root”);

重新登录创建 用户mysql

./bin/mysql -P3336 -uroot –socket=/data/3336/mysql.sock -proot

grant all PRIVILEGES on . to ‘mysql‘@’%’ identified by “mysql” with GRANT option ;

flush privileges;

从5.7版本开始就要安装包安装cmake3
